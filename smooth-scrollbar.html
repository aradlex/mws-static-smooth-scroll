🚀 Get Started with Smooth Scrollbar

Ready to implement custom scrollbars? Here's everything you need to get started.

🚀 Quick Start

// Load Smooth Scrollbar
<script src="libs/smooth-scrollbar.min.js"></script>

⚙️ Basic Setup

const scrollbar = Scrollbar.init(document.body, {
  damping: 0.1,
  thumbMinSize: 20,
  renderByPixels: true,
  alwaysShowTracks: false,
  continuousScrolling: true
});

// Custom styling
scrollbar.track.xAxis.element.style.background = '#f0f0f0';
scrollbar.track.yAxis.element.style.background = '#f0f0f0';
