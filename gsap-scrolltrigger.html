🚀 Get Started with GSAP ScrollTrigger

Ready to create professional scroll animations? Here's everything you need to get started.

🚀 Quick Start

// Load GSAP and ScrollTrigger
<script src="libs/gsap.min.js"></script>
<script src="libs/ScrollTrigger.min.js"></script>

⚙️ Basic Animation

gsap.registerPlugin(ScrollTrigger);

gsap.fromTo(".fade-in", 
  { opacity: 0, y: 50 },
  {
    opacity: 1,
    y: 0,
    duration: 1,
    scrollTrigger: {
      trigger: ".fade-in",
      start: "top 80%",
      end: "bottom 20%"
    }
  }
);
